cmake_minimum_required(VERSION 3.19)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../shared/product-version.txt" ZTL_VERSION LIMIT_COUNT 1)
project(ZTLServerManager VERSION ${ZTL_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INSTALL_BINDIR "." CACHE PATH "Z.T.L executables and runtime DLLs" FORCE)
set(CMAKE_INSTALL_LIBDIR "." CACHE PATH "Z.T.L libraries" FORCE)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Widgets)
qt_standard_project_setup()
qt_add_library(ztl_manager STATIC app/managerwindow.cpp app/managerwindow.h app/bootstrapprocess.cpp app/bootstrapprocess.h)
target_include_directories(ztl_manager PUBLIC app)
target_link_libraries(ztl_manager PUBLIC Qt6::Core Qt6::Widgets)
qt_add_executable(ZTLServerManager WIN32 app/main.cpp)
set_target_properties(ZTLServerManager PROPERTIES OUTPUT_NAME "Z.T.L.ServerManager")
target_link_libraries(ZTLServerManager PRIVATE ztl_manager)
find_program(ZTL_POWERSHELL NAMES pwsh powershell PATHS "$ENV{SystemRoot}/System32/WindowsPowerShell/v1.0" REQUIRED)
add_custom_target(manager_runtime
 COMMAND "${ZTL_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/prepare_runtime.ps1" -Destination "$<TARGET_FILE_DIR:ZTLServerManager>/runtime"
 COMMAND "${ZTL_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_runtime.ps1" -Destination "$<TARGET_FILE_DIR:ZTLServerManager>/runtime"
 COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../shared/release-layout.json" "$<TARGET_FILE_DIR:ZTLServerManager>/scripts/release-layout.json"
 COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/scripts" "$<TARGET_FILE_DIR:ZTLServerManager>/scripts"
 VERBATIM)
add_dependencies(ZTLServerManager manager_runtime)
install(TARGETS ZTLServerManager RUNTIME DESTINATION .)
install(FILES ../README.md ../instruct.md DESTINATION .)
install(FILES ../shared/release-layout.json DESTINATION scripts)
install(FILES scripts/bootstrap_worker.py scripts/bootstrap_remote.py scripts/release_bundle.py scripts/database_restore.py scripts/release_switch.py scripts/remote_operation.py DESTINATION scripts)
install(DIRECTORY "$<TARGET_FILE_DIR:ZTLServerManager>/runtime/" DESTINATION runtime)
qt_generate_deploy_app_script(TARGET ZTLServerManager OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 qt_add_executable(ztl_manager_tests tests/test_manager.cpp)
 target_link_libraries(ztl_manager_tests PRIVATE ztl_manager Qt6::Test)
 target_compile_definitions(ztl_manager_tests PRIVATE ZTL_MANAGER_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests")
 add_dependencies(ztl_manager_tests manager_runtime)
 add_test(NAME manager COMMAND ztl_manager_tests)
 set_tests_properties(manager PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
endif()

target_compile_definitions(ZTLServerManager PRIVATE ZTL_PRODUCT_VERSION="${PROJECT_VERSION}")
